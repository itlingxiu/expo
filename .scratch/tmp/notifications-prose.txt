160|  let token;

171|  const { status: existingStatus } = await Notifications.getPermissionsAsync();
172|  let finalStatus = existingStatus;
173|  if (existingStatus !== 'granted') {
174|    const { status } = await Notifications.requestPermissionsAsync();
175|    finalStatus = status;
176|  }
177|  if (finalStatus !== 'granted') {
178|    alert('Failed to get push token for push notification!');
179|    return;
180|  }
181|  // Learn more about projectId:
182|  // https://docs.expo.dev/push-notifications/push-notifications-setup/#configure-projectid
183|  // EAS projectId is used here.
184|  try {
185|    const projectId =
186|      Constants?.expoConfig?.extra?.eas?.projectId ?? Constants?.easConfig?.projectId;
187|    if (!projectId) {
188|      throw new Error('Project ID not found');

192|        projectId,
193|      })

231|function useNotificationObserver() {
232|  useEffect(() => {
233|    function redirect(notification: Notifications.Notification) {
234|      const url = notification.request.content.data?.url;
235|      if (typeof url === 'string') {
236|        /* @info Push the URL. You may want to verify the format before navigating. */
237|        router.push(url);
238|        /* @end */
239|        /* @info Clear the stored response so a layout re-mount does not redirect to the same URL again. */
240|        Notifications.clearLastNotificationResponse();
241|        /* @end */
242|      }
243|    }
245|    /* @info Handle the initial push notification. */
246|    const response = Notifications.getLastNotificationResponse();
247|    if (response?.notification) {
248|      redirect(response.notification);
249|    }
251|    /* @info Listen for runtime notifications. */
252|    const subscription = Notifications.addNotificationResponseReceivedListener(response => {
253|      /* @end */
254|      redirect(response.notification);
255|    });
257|    return () => {
258|      subscription.remove();
259|    };
260|  }, []);
